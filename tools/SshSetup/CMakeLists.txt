cmake_minimum_required(VERSION 3.21)
project(OrbiDevSshSetup VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Widgets Network)
get_target_property(setup_qmake Qt5::qmake IMPORTED_LOCATION)
get_filename_component(setup_qt_bin "${setup_qmake}" DIRECTORY)
function(setup_target target)
    target_compile_features(${target} PRIVATE cxx_std_20)
    set_target_properties(${target} PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF
        VS_DEBUGGER_ENVIRONMENT "PATH=${setup_qt_bin};%PATH%")
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /analyze /analyze:external-)
endfunction()
add_library(SshSetupCore STATIC IdentityService.cpp IdentityService.h
    AuthorizationService.cpp AuthorizationService.h EnvironmentService.cpp EnvironmentService.h)
target_include_directories(SshSetupCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(SshSetupCore PUBLIC Qt5::Core PRIVATE Qt5::Network advapi32)
setup_target(SshSetupCore)
add_library(SshSetupUi STATIC SetupWindow.cpp SetupWindow.h SetupTheme.cpp SetupTheme.h)
target_link_libraries(SshSetupUi PUBLIC SshSetupCore Qt5::Widgets PRIVATE shell32)
setup_target(SshSetupUi)
foreach(app IN ITEMS OrbiDevSetup OrbiDevHomeSetup)
    add_executable(${app} WIN32 main.cpp)
    target_compile_definitions(${app} PRIVATE ORBIDEV_VERSION="${PROJECT_VERSION}-preview")
    target_link_libraries(${app} PRIVATE SshSetupUi)
    setup_target(${app})
endforeach()
target_compile_definitions(OrbiDevHomeSetup PRIVATE ORBIDEV_HOME_SETUP)
if(BUILD_TESTING)
    find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Test)
    foreach(suite IN ITEMS IdentityService AuthorizationService EnvironmentService)
        add_executable(${suite}Tests ${suite}Tests.cpp)
        target_link_libraries(${suite}Tests PRIVATE SshSetupCore Qt5::Test advapi32)
        setup_target(${suite}Tests)
        add_test(NAME setup.${suite} COMMAND ${suite}Tests -o -,txt)
        set_tests_properties(setup.${suite} PROPERTIES ENVIRONMENT "PATH=${setup_qt_bin}\;$ENV{PATH}" TIMEOUT 90)
    endforeach()
    add_executable(SetupWindowTests SetupWindowTests.cpp)
    target_link_libraries(SetupWindowTests PRIVATE SshSetupUi Qt5::Test)
    setup_target(SetupWindowTests)
    add_test(NAME setup.ui COMMAND SetupWindowTests -o -,txt)
    set_tests_properties(setup.ui PROPERTIES ENVIRONMENT "PATH=${setup_qt_bin}\;$ENV{PATH}" TIMEOUT 90)
endif()
