cmake_minimum_required(VERSION 3.21)
project(RADManager VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RADManager requires Windows, MSVC and x64")
endif()

option(RAD_ENABLE_ANALYSIS "启用 MSVC 静态分析" OFF)
option(RAD_BUILD_TERMINAL_POC "构建 M0-02 Qt 离线终端演示" ON)
option(RAD_BUILD_CONPTY_POC "构建 M0-01 ConPTY 演示及回归测试" ON)
option(RAD_BUILD_APPLICATIONS "构建 M1 Client/Agent 工程骨架" ON)
include(CTest)
if(RAD_BUILD_CONPTY_POC)
include(cmake/ConPtyRuntime.cmake)
option(RAD_TEST_SYSTEM_CONPTY "将原生系统 ConPTY 诊断加入 CTest（本机已知句柄增长）" OFF)

add_library(RADConPtyPoc STATIC platforms/windows/ConPty/ConPtySession.cpp)
target_include_directories(RADConPtyPoc PUBLIC platforms/windows/ConPty)
add_executable(FakeCli tests/fixtures/FakeCli/main.cpp)
add_executable(ConPtyPoc tests/integration/ConPtyPocTests.cpp)
add_executable(ConPtyOsProbe tools/ConPtyOsProbe.cpp)
add_executable(ConPtyRuntimeProbe tools/ConPtyOsProbe.cpp)
target_compile_definitions(ConPtyRuntimeProbe PRIVATE RAD_CONPTY_RUNTIME)
target_link_libraries(RADConPtyPoc PRIVATE RADConPtyRuntime)
target_link_libraries(ConPtyRuntimeProbe PRIVATE RADConPtyRuntime)
rad_deploy_conpty(ConPtyPoc)
# 两个程序共用输出目录，由同一部署目标复制，避免并行写同一文件。
add_dependencies(ConPtyRuntimeProbe ConPtyPocRuntime)
target_link_libraries(ConPtyPoc PRIVATE RADConPtyPoc)
add_dependencies(ConPtyPoc FakeCli)

file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/InvalidProgram.exe" CONTENT "M0 invalid executable fixture")

foreach(target RADConPtyPoc FakeCli ConPtyPoc ConPtyOsProbe ConPtyRuntimeProbe)
    target_compile_features(${target} PRIVATE cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    if(RAD_ENABLE_ANALYSIS)
        target_compile_options(${target} PRIVATE /analyze)
    endif()
endforeach()

if(BUILD_TESTING)
    foreach(scenario interaction throughput shutdown validation lifecycle)
        add_test(NAME conpty.${scenario} COMMAND ConPtyPoc ${scenario})
        set_tests_properties(conpty.${scenario} PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
    endforeach()
    add_test(NAME conpty.runtime-handles COMMAND ConPtyRuntimeProbe)
    set_tests_properties(conpty.runtime-handles PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    if(RAD_TEST_SYSTEM_CONPTY)
        add_test(NAME conpty.os-handles COMMAND ConPtyOsProbe)
        set_tests_properties(conpty.os-handles PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS diagnostic)
    endif()
endif()

endif()

if(RAD_BUILD_TERMINAL_POC OR RAD_BUILD_APPLICATIONS)
    find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel)
    get_target_property(rad_qmake Qt5::qmake IMPORTED_LOCATION)
    get_filename_component(RAD_QT_BIN "${rad_qmake}" DIRECTORY)
endif()

if(RAD_BUILD_TERMINAL_POC)
    add_subdirectory(tools/TerminalPoc)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT TerminalPoc)
endif()

if(RAD_BUILD_APPLICATIONS)
    find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Sql)
    function(rad_configure_target target)
        target_compile_features(${target} PRIVATE cxx_std_20)
        set_target_properties(${target} PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF
            VS_DEBUGGER_ENVIRONMENT "PATH=${RAD_QT_BIN};%PATH%")
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
        if(RAD_ENABLE_ANALYSIS)
            target_compile_options(${target} PRIVATE /analyze /analyze:external-)
        endif()
    endfunction()
    add_subdirectory(core)
    add_subdirectory(apps/RemoteClient)
    add_subdirectory(apps/RemoteAgent)
    if(BUILD_TESTING)
        find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Test)
        add_subdirectory(tests/unit)
    endif()
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT RemoteClient)
endif()
